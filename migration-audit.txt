    FROM unnest(c.conkey) WITH ORDINALITY AS u(attnum, ord)

    SELECT string_agg(a.attname, ', ' ORDER BY u.ord)

  pg_get_constraintdef(c.oid)                  AS constraint_definition,
  c.deferrable,

3. PRIVILEGIOS EXECUTE DAS 4 FUNCOES BIOMETRICAS

-- PRE-CONDIÃ‡ÃƒO: cada funcao pode ou nao existir. Confirmar SEPARADAMENTE
--   (to_regprocedure para funcoes com assinatura):
--     SELECT to_regprocedure('public.user_can_use_biometrics(uuid)')        IS NOT NULL;
--     SELECT to_regprocedure('public.student_has_active_biometrics(uuid)')  IS NOT NULL;
--     SELECT to_regprocedure('public.my_biometric_enrollments()')           IS NOT NULL;
--     SELECT to_regprocedure('public.guard_biometric_identity()')          IS NOT NULL;
--
-- TRES ASPECTOS DISTINTOS:
--   A) GRANT direto a um role nomeado: visivel em routine_privileges.
--   B) GRANT a PUBLIC: visivel em aclexplode sobre proacl/acldefault
--      quando ae.grantee = 0.
--   C) Membership em outro role: NAO visivel nesta secao.
--      Se role R tem EXECUTE e role S e MEMBER de R, S executa sem
--      que isso apareca. Verificar via pg_auth_members (superuser).
--   ACLs NAO resolvem membership em roles.
--
-- PERMISSAO: acesso a information_schema e pg_proc.

-- 3.1 Grants diretos
SELECT
  r.routine_schema,
  r.routine_name,
  r.specific_name,
  rp.grantee,
  rp.privilege_type,
  rp.is_grantable
FROM information_schema.routines r
LEFT JOIN information_schema.routine_privileges rp
       ON rp.specific_schema = r.routine_schema
      AND rp.specific_name   = r.specific_name
WHERE r.routine_schema = 'public'
  AND r.routine_name IN (
    'user_can_use_biometrics',
    'student_has_active_biometrics',
    'my_biometric_enrollments',
    'guard_biometric_identity'
  )
ORDER BY r.routine_name, rp.grantee NULLS LAST;

-- 3.2 ACLs via pg_proc.proacl + acldefault para proacl NULL
SELECT
  p.proname                                      AS function_name,
  p.pronamespace::regnamespace                  AS function_schema,
  format('%s(%s)', p.proname, pg_get_function_arguments(p.oid))
                                                   AS signature,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE rr.rolname END
                                                   AS grantee_name,
  ae.privilege_type,
  ae.is_grantable,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE 'role' END
                                                   AS grant_type,
  CASE WHEN p.proacl IS NOT NULL THEN 'explicit_proacl' ELSE 'acldefault' END
                                                   AS acl_source
FROM pg_proc p
JOIN pg_namespace n ON n.oid = p.pronamespace
CROSS JOIN LATERAL aclexplode(
  COALESCE(p.proacl, acldefault('f', p.proowner))
) AS ae
LEFT JOIN pg_roles rr ON rr.oid = ae.grantee
WHERE n.nspname = 'public'
  AND p.proname IN (
    'user_can_use_biometrics',
    'student_has_active_biometrics',
    'my_biometric_enrollments',
    'guard_biometric_identity'
  )
ORDER BY p.proname, ae.grantee, ae.privilege_type;

DO $$
DECLARE
  v_has_ref    boolean;
  v_has_status boolean;
  v_has_score  boolean;
  v_session_exists boolean;
BEGIN
  SELECT EXISTS (
    SELECT 1 FROM information_schema.tables
    WHERE table_schema = 'public' AND table_name = 'exam_sessions'
  ) INTO v_session_exists;

  IF NOT v_session_exists THEN
    RAISE NOTICE 'exam_sessions: tabela nao existe neste banco';
    RETURN;
  END IF;

  SELECT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'exam_sessions'
      AND column_name = 'biometric_ref'
  ) INTO v_has_ref;

  SELECT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'exam_sessions'
      AND column_name = 'biometric_status'
  ) INTO v_has_status;

  SELECT EXISTS (
    SELECT 1 FROM information_schema.columns
    WHERE table_schema = 'public' AND table_name = 'exam_sessions'
      AND column_name = 'biometric_match_score'
  ) INTO v_has_score;

  IF v_has_ref THEN
    RAISE NOTICE 'biometric_ref NOT NULL: %',
      (SELECT COUNT(*) FROM public.exam_sessions WHERE biometric_ref IS NOT NULL);
  END IF;

  IF v_has_status THEN
    RAISE NOTICE 'biometric_status NOT NULL: %',
      (SELECT COUNT(*) FROM public.exam_sessions WHERE biometric_status IS NOT NULL);
  END IF;

  IF v_has_score THEN
    RAISE NOTICE 'biometric_match_score NOT NULL: %',
      (SELECT COUNT(*) FROM public.exam_sessions WHERE biometric_match_score IS NOT NULL);
  END IF;

  IF NOT (v_has_ref OR v_has_status OR v_has_score) THEN
    RAISE NOTICE 'Nenhuma coluna biometrica existe em exam_sessions neste banco';
  END IF;
END $$;

--   B) PRIVILEGIOS A PUBLIC: visiveis em aclexplode sobre relacl/acldefault
--      quando ae.grantee = 0.
--   C) OWNERSHIP: identificado por pg_class.relowner. O owner tem todos
--      os privilegios automaticamente â€” NAO aparece em table_privileges.
--   D) ACL PADRAO: se relacl for NULL, a ACL efetiva e
--      acldefault('r', relowner).
--   E) MEMBERSHIP em roles: NAO visivel nesta secao. Verificar via
--      pg_auth_members (superuser).

-- table_privileges sozinha NAO cobre todos os privilegios efetivos.

-- 7.1A PrivilÃ©gios listados por information_schema.table_privileges
-- A view pode listar PUBLIC quando aplicavel. Roles com privilegios
-- herdados via membership NAO aparecem nesta view.
SELECT
  tp.table_schema,
  tp.table_name,
  tp.grantee,
  tp.privilege_type,
  tp.is_grantable
FROM information_schema.table_privileges tp
WHERE tp.table_schema = 'public'
  AND tp.table_name IN (
    'user_biometrics',
    'biometric_audit_log'
  )
ORDER BY tp.table_name, tp.grantee, tp.privilege_type;

-- 7.1B Ownership das tabelas
SELECT
  c.relnamespace::regnamespace                  AS table_schema,
  c.relname                                      AS table_name,
  c.relowner::regrole                          AS owner
FROM pg_class c
WHERE c.relnamespace = 'public'::regnamespace
  AND c.relname IN (
    'user_biometrics',
    'biometric_audit_log'
  )
ORDER BY c.relname;

-- 7.2 ACL via aclexplode sobre relacl/acldefault
-- user_biometrics â€” so executar se existir
SELECT
  c.relname                                      AS table_name,
  c.relowner::regrole                          AS table_owner,
  c.relacl IS NOT NULL                          AS has_explicit_relacl,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE ae.grantee::regrole::text END
                                                   AS grantee,
  ae.privilege_type,
  ae.is_grantable,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE 'role' END
                                                   AS grant_type,
  CASE WHEN c.relacl IS NOT NULL THEN 'from_relacl' ELSE 'from_acldefault' END
                                                   AS acl_source
FROM pg_class c
CROSS JOIN LATERAL aclexplode(
  COALESCE(c.relacl, acldefault('r', c.relowner))
) AS ae
WHERE c.relnamespace = 'public'::regnamespace
  AND c.relname = 'user_biometrics';

-- biometric_audit_log â€” so executar se existir
SELECT
  c.relname                                      AS table_name,
  c.relowner::regrole                          AS table_owner,
  c.relacl IS NOT NULL                          AS has_explicit_relacl,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE ae.grantee::regrole::text END
                                                   AS grantee,
  ae.privilege_type,
  ae.is_grantable,
  CASE WHEN ae.grantee = 0 THEN 'PUBLIC' ELSE 'role' END
                                                   AS grant_type,
  CASE WHEN c.relacl IS NOT NULL THEN 'from_relacl' ELSE 'from_acldefault' END
                                                   AS acl_source
FROM pg_class c
CROSS JOIN LATERAL aclexplode(
  COALESCE(c.relacl, acldefault('r', c.relowner))
) AS ae
WHERE c.relnamespace = 'public'::regnamespace
  AND c.relname = 'biometric_audit_log';

-- 7.3 Privilegios padrao de novos objetos no schema public
-- pg_default_acl trata de privilegios PADRAO de objetos FUTUROS.
-- NAO e a ACL de tabelas ja existentes â€” ver secao 7.2.
-- Requer superuser para resultado completo.
SELECT
  n.nspname                                      AS schema_name,
  a.defaclrole::regrole                         AS default_for_role,
  CASE a.defaclobjtype
    WHEN 'r' THEN 'tables'
    WHEN 'f' THEN 'functions'
    WHEN 'T' THEN 'types'
    WHEN 'n' THEN 'schemas'
    ELSE a.defaclobjtype
  END                                            AS object_type,
  a.defaclacl                                    AS default_privileges
FROM pg_default_acl a
JOIN pg_namespace n ON n.oid = a.defaclnamespace
WHERE n.nspname = 'public';
